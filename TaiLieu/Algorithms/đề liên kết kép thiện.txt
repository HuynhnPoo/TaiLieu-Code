- tên hàng
- số lượng
- đơn giá

Đọc từ file mathang.txt
In danh sách
Tìm mặt hàng có số lượng nhỏ nhất

#include<iostream>
#include<fstream>

using namespace std;

//=== CAU TRUC ===
struct MatHang
{
    string tenHang;
    int soLuong;
    float donGia;
};

struct Node
{
    MatHang data;
    Node *prev;
    Node *next;
};

struct DSLKK
{
    Node *head;
    Node *tail;
};

//=== HAM ===
void khoiTao(DSLKK &ds);
Node* taoNode(MatHang mh);
void themCuoi(DSLKK &ds, MatHang mh);
void hienDS(DSLKK ds);
MatHang timSoLuongMatHang(DSLKK ds);

//=== MAIN ===
int main()
{
    ifstream fin("mathang.txt");

    DSLKK ds;
    khoiTao(ds);

    int n;
    fin >> n;
    fin.ignore();

    for(int i = 0; i < n; i++)
    {
        MatHang mh;

        getline(fin, mh.tenHang);
        fin >> mh.soLuong;
        fin >> mh.donGia;
        fin.ignore();

        themCuoi(ds, mh);
    }

    cout << "DANH SACH MAT HANG:\n";
    hienDS(ds);

    MatHang matHang = timSoLuongMatHang(ds);

    cout << "\nMAT HANG CO SO LUONG NHO NHAT:\n";
    cout << matHang.tenHang << endl;
    cout << matHang.soLuong << endl;
    cout << matHang.donGia << endl;

    return 0;
}

//=== HAM XU LY ===

void khoiTao(DSLKK &ds)
{
    ds.head = NULL;
    ds.tail = NULL;
}

Node* taoNode(MatHang mh)
{
    Node *p = new Node;
    p->data = mh;
    p->prev = NULL;
    p->next = NULL;
    return p;
}

void themCuoi(DSLKK &ds, MatHang mh)
{
    Node *p = taoNode(mh);

    if(ds.head == NULL)
    {
        ds.head = ds.tail = p;
    }
    else
    {
        ds.tail->next = p;
        p->prev = ds.tail;
        ds.tail = p;
    }
}

void hienDS(DSLKK ds)
{
    Node *p = ds.head;

    while(p != NULL)
    {
        cout << p->data.tenHang << endl;
        cout << p->data.soLuong << endl;
        cout << p->data.donGia << endl;
        cout << endl;

        p = p->next;
    }
}

MatHang timSoLuongMatHang(DSLKK ds)
{
    Node *p = ds.head;

    MatHang matHang = p->data;
    p = p->next;

    while(p != NULL)
    {
        if(p->data.soLuong < matHang.soLuong)
            matHang = p->data;

        p = p->next;
    }

    return matHang;
}
