#include<iostream>
#include<fstream>
#include<stdio.h>

using namespace std;

//Cau truc mat hang
struct MatHang
{
    string tenHang;
    int soLuong;
    float donGia;
};

//Node
struct Node
{
    MatHang data;
    Node *prev;
    Node *next;
};

//DSLKK
struct DSLKK
{
    Node *head;
    Node *tail;
};

//Khai bao ham
void khoiTao(DSLKK &ds);
Node* taoNode(MatHang mh);
void themCuoi(DSLKK &ds,MatHang mh);
void hienDS(DSLKK ds);
MatHang timMin(DSLKK ds);

//===MAIN===
int main()
{
    ifstream fin("mathang.txt");

    DSLKK ds;
    khoiTao(ds);

    int n;
    fin>>n;
    fin.ignore();

    for(int i=0;i<n;i++)
    {
        MatHang mh;

        getline(fin,mh.tenHang);
        fin>>mh.soLuong;
        fin>>mh.donGia;
        fin.ignore();

        themCuoi(ds,mh);
    }

    cout<<"Danh sach mat hang:\n";
    hienDS(ds);

    MatHang mhMin = timMin(ds);

    cout<<"\nMat hang co so luong nho nhat:\n";
    cout<<mhMin.tenHang<<endl;
    cout<<mhMin.soLuong<<endl;
    cout<<mhMin.donGia<<endl;

    return 0;
}

//===HAM===

void khoiTao(DSLKK &ds)
{
    ds.head = NULL;
    ds.tail = NULL;
}

Node* taoNode(MatHang mh)
{
    Node *p = new Node;
    p->data = mh;
    p->prev = NULL;
    p->next = NULL;
    return p;
}

void themCuoi(DSLKK &ds,MatHang mh)
{
    Node *p = taoNode(mh);

    if(ds.head == NULL)
    {
        ds.head = ds.tail = p;
    }
    else
    {
        ds.tail->next = p;
        p->prev = ds.tail;
        ds.tail = p;
    }
}

void hienDS(DSLKK ds)
{
    Node *p = ds.head;

    while(p != NULL)
    {
        cout<<p->data.tenHang<<endl;
        cout<<p->data.soLuong<<endl;
        cout<<p->data.donGia<<endl;
        cout<<endl;

        p = p->next;
    }
}

MatHang timMin(DSLKK ds)
{
    Node *p = ds.head;

    MatHang mhMin = p->data;
    p = p->next;

    while(p != NULL)
    {
        if(p->data.soLuong < mhMin.soLuong)
            mhMin = p->data;

        p = p->next;
    }

    return mhMin;
}
