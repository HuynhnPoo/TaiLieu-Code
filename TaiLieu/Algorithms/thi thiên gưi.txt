Mỗi tỉnh có thông tin: tên tỉnh, diện tích, dân số.
Danh sách n tỉnh đọc từ tệp văn bản 'tinh.txt'.
Đưa ra màn hình danh sách ban đầu và sau khi sắp xếp.

#include<iostream>
#include<fstream>
#include<stdio.h>

using namespace std;

//Khai bao cau truc tinh/thanh pho
struct Tinh
{
    string tenTinh;
    float dienTich;
    int danSo;
};

//Khai bao ham
void hienDS(Tinh a[],int n);
void quickSort(Tinh a[],int left,int right);

//===chuong trinh chinh===
int main()
{
    //Khai bao tep vao
    ifstream fin("tinh.txt");

    //Khai bao bien
    int n;

    //Doc so luong tinh
    fin>>n;
    fin.ignore();

    //Tao mang dong
    Tinh *a = new Tinh[n];

    //Doc du lieu tu tep vao mang
    for(int i=0;i<n;i++)
    {
        getline(fin,a[i].tenTinh);
        fin>>a[i].dienTich;
        fin>>a[i].danSo;
        fin.ignore();
    }

    //Dua ra danh sach ban dau
    cout<<"Danh sach ban dau:\n";
    hienDS(a,n);

    //Sap xep nhanh theo dan so giam dan
    quickSort(a,0,n-1);

    //Dua ra danh sach sau sap xep
    cout<<"\nDanh sach sau khi sap xep:\n";
    hienDS(a,n);

    cout<<endl;
    return 0;
}

//===dinh nghia ham===

//Ham hien thi danh sach
void hienDS(Tinh a[],int n)
{
    for(int i=0;i<n;i++)
    {
        cout<<a[i].tenTinh<<endl;
        cout<<a[i].dienTich<<endl;
        cout<<a[i].danSo<<endl;
        cout<<endl;
    }
}

//Ham sap xep nhanh
void quickSort(Tinh a[],int left,int right)
{
    int i = left;
    int j = right;

    //Chon khoa giua
    int x = a[(left+right)/2].danSo;

    //Phan doan
    while(i<=j)
    {
        while(a[i].danSo > x) i++;
        while(a[j].danSo < x) j--;

        if(i<=j)
        {
            //Hoan vi
            Tinh tg = a[i];
            a[i] = a[j];
            a[j] = tg;

            i++;
            j--;
        }
    }

    //De quy
    if(left<j)
        quickSort(a,left,j);

    if(i<right)
        quickSort(a,i,right);
}
