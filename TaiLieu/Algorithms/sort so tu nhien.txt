void hienThi(int a[], int n) {
    for (int i = 0; i < n; i++) {

void Swap(int &a,int &b)
{
    int temp =a;
    a=b;
    b=temp;
}

void SelectionSort(int a[],int n){
    for(int i =0; i<n ;i++){
        int min=i;
        cout<<"min = "<<min<<    endl;
        for(int j =i+1;j<n;j++){
            cout<<"j = "<<j<<endl;
            if(a[j]<a[min]){
                min = j;
            }   
        }
    swap(a[i],a[min]);
    }
}

void bubbleSort(int a[],int n)
{
    for(int i=0;i<n-1;i++)   
        {
            for(int j=0;j<n-i-1;j++){
                cout<<"j = "<<j<< " " << n-i-1 <<" "<< a[j+1] << endl;
                if(a[j]>a[j+1])
                    swap(a[j],a[j+1]);
            }
        }
}

void insertionSort(int a[],int n)
{
   for(int i=1;i<n;i++){
       int key=a[i], pos=i-1;
       while(pos>=0 && a[pos]> key){
           a[pos+1] =a[pos];
           pos--;    
       }
       a[pos+1]=key;
   }
}

void quickSort(int a[],int left,int right)
{
    if(left >= right) return;
    int i = left, j = right,pivot=a[(left +right)/2];

    while(i<=j){
       while(a[i]<pivot) i++;
        while(a[j]>pivot) j--;
        if(i<=j)
        {
            Swap(a[i],a[j]);
            i++;    
            j--;
        }
    }

    if(left<j) quickSort(a,left,j);
    if(right>i) quickSort(a,i,right);
}

void heapify(int a[], int n, int i)
{
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if(left < n && a[left] > a[largest])
        largest = left;

    if(right < n && a[right] > a[largest])
        largest = right;

    if(largest != i)
    {
        Swap(a[i], a[largest]);
        heapify(a, n, largest);
    }
}

void heapSort(int a[], int n)
{
    for(int i = n / 2 - 1; i >= 0; i--)
        heapify(a, n, i);

    for(int i = n - 1; i > 0; i--)
    {
        Swap(a[0], a[i]);
        heapify(a, i, 0); 
    }
}
